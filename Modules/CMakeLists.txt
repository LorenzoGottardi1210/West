# 1. geenrating west_version.f90 from VERSION.json using read_json.py 
execute_process(
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/Pytools/read_json.py ${CMAKE_SOURCE_DIR}/VERSION.json version
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    OUTPUT_VARIABLE WEST_VERSION
    OUTPUT_STRIP_TRAILING_WHITESPACE
)

message(STATUS "WEST Modules: detecting version -> ${WEST_VERSION}")

execute_process(
    COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/update_west_version ${CMAKE_SOURCE_DIR} ${WEST_VERSION}
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    RESULT_VARIABLE UPDATE_VERSION_RESULT
)

if(NOT UPDATE_VERSION_RESULT EQUAL 0)
    message(FATAL_ERROR "Error during generation of west_version.f90 in Modules")
endif()

# 2. creating the target west_modules as a static library
add_library(west_modules STATIC 
    west_gpu.f90 
    west_version.f90 
    westcom.f90
)

# 3. target proprieties setting to enable preprocessing for Fortran files (to handle GPU directives) 
set_target_properties(west_modules PROPERTIES Fortran_PREPROCESS ON)

# 4. exporting the modules generated by this library to the next steps 
target_include_directories(west_modules PUBLIC ${CMAKE_CURRENT_BINARY_DIR})

# 5. linking the libraries needed by west_modules, including the Quantum ESPRESSO libraries
target_link_libraries(west_modules PUBLIC 
    forpy 
    qe_modules 
    qe_fftx 
    qe_pw 
    qe_utilx 
    qe_upflib
)
